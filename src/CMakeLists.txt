SUBDIRLIST(SUBDIRS ${CMAKE_CURRENT_SOURCE_DIR})

FOREACH(subdir ${SUBDIRS})
	set(filename "${CMAKE_CURRENT_SOURCE_DIR}/${subdir}/arith.c")
    add_executable(arith_${subdir} ${filename})
	add_test(arith_${subdir} arith_${subdir})
	if (benchmark_FOUND)
		set(filename "${CMAKE_CURRENT_SOURCE_DIR}/${subdir}/bench.cpp")
		add_executable(bench_${subdir} ${filename})
		target_link_libraries(bench_${subdir} benchmark::benchmark)
		add_test(bench_${subdir} bench_${subdir})
	endif()
ENDFOREACH()

# CUDA implementations: test_cuda_<field> (googletest), bench_cuda_<field> (nvbench)
if(FQ_ARITHMETIC_CUDA AND CMAKE_CUDA_COMPILER)
	FOREACH(subdir ${SUBDIRS})
		set(filename "${CMAKE_CURRENT_SOURCE_DIR}/${subdir}/test_cuda.cu")
		if (EXISTS ${filename})
			add_executable(test_cuda_${subdir} ${filename})
			target_link_libraries(test_cuda_${subdir} GTest::gtest_main)
			add_test(test_cuda_${subdir} test_cuda_${subdir})
		endif()

		set(filename "${CMAKE_CURRENT_SOURCE_DIR}/${subdir}/bench_cuda.cu")
		if (EXISTS ${filename})
			add_executable(bench_cuda_${subdir} ${filename})
			target_link_libraries(bench_cuda_${subdir} nvbench::main)
		endif()
	ENDFOREACH()
endif()
